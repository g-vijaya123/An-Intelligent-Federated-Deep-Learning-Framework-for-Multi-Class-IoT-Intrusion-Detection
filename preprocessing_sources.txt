def stable_hash_fraction(
    dataset_name,
    source_file,
    row_number,
    label,
    stage,
):
    """
    Deterministic pseudo-random number in [0, 1).

    Including dataset, source file, row number and label makes
    selection reproducible without loading the full dataset
    into memory.
    """

    text = (
        f"{SEED}|"
        f"{stage}|"
        f"{dataset_name}|"
        f"{source_file}|"
        f"{row_number}|"
        f"{label}"
    )

    digest = hashlib.blake2b(
        text.encode("utf-8"),
        digest_size=8,
    ).digest()

    value = int.from_bytes(
        digest,
        byteorder="big",
        signed=False,
    )

    return value / float(2**64)


def append_dataframe(
    dataframe,
    output_file,
    write_header,
):
    if dataframe.empty:
        return write_header

    dataframe.to_csv(
        output_file,
        mode="a",
        index=False,
        header=write_header,
    )

    return False

for dataset_name, pattern in DATASETS.items():

    section(f"PROCESSING {dataset_name}")

    files = sorted(

    )

    if not files:
        raise FileNotFoundError(

    print(
        f"Input chunks found: {len(files)}"
    )

    client_dir = (

    test_header = True
    selected_header = True

    original_counts = {

    selected_counts = {
        class_name: 0

    }

    dataset_original = 0
    dataset_selected = 0

    # --------------------------------------------------------

        files,
        start=1,

        print(
            f"\n[{file_index}/{len(files)}] "
            f"{input_file.name}"
        )

        file_row_offset = 0

        for chunk_number, chunk in enumerate(
            pd.read_csv(
                input_file,
                chunksize=READ_CHUNK_SIZE,
                low_memory=False,

            required_columns = (
                FEATURES

                    "label",

                ]
            )

            missing = [
                column
                for column in required_columns
                if column not in chunk.columns
            ]

            if missing:
                raise ValueError(
                    f"Missing columns in "
                    f"{input_file.name}: {missing}"
                )

            # -----------------------------------------------
            # Keep only expected labels
            # -----------------------------------------------

            chunk["label"] = (
                chunk["label"]
                .astype(str)
                .str.strip()
            )

            unexpected = sorted(
                set(chunk["label"].unique())

            )

            if unexpected:
                raise ValueError(
                    f"Unexpected labels found in "
                    f"{input_file.name}: "
                    f"{unexpected}"
                )

            current_rows = len(chunk)

            dataset_original += current_rows

            # -----------------------------------------------
            # Original class counts
            # -----------------------------------------------

            original_value_counts = (
                chunk["label"]
                .value_counts()
            )

            for class_name, count in (
                original_value_counts.items()

                original_counts[class_name] += int(
                    count
                )

            # -----------------------------------------------
            # Deterministic 20% class-aware sampling
            # -----------------------------------------------

            selected_mask = np.zeros(
                current_rows,
                dtype=bool,

            labels = (
                chunk["label"]
                .to_numpy()
            )

            for local_index in range(
                current_rows

                global_row_number = (
                    file_row_offset
                    + local_index

                    label=label,
                    stage="sample20",

                    selected_mask[
                        local_index
                    ] = True

            selected = (
                chunk.loc[

                ]

            )

            file_row_offset += current_rows

            if selected.empty:

                    f"  chunk {chunk_number}: "
                    f"{current_rows:,} read, "

            # Label IDs

                    class_name
                ] += int(count)

            )

            # -----------------------------------------------
            # 80/20 train/test split WITHIN selected subset
            # -----------------------------------------------

            train_mask = np.zeros(
                len(selected),
                dtype=bool,
            )

            selected_labels = (
                selected["label"]
                .to_numpy()
            )

                .to_numpy()
            )

                len(selected)

                    selected_labels[
                        position

                    selected_indices[

                    source_file=input_file.name,
                    row_number=original_index,
                    label=label,
                    stage="train_test",
                )

                    train_mask[
                        position
                    ] = True

            train_part = (
                selected.iloc[
                    train_mask
                ]
                .copy()
            )

            test_part = (
                selected.iloc[
                    ~train_mask
                ]
                .copy()
            )

            # -----------------------------------------------
            # Save train
            # -----------------------------------------------

            if not train_part.empty:

                train_header = append_dataframe(

                    train_file,

def numeric(series):
    return (
        pd.to_numeric(series, errors="coerce")
        .replace([np.inf, -np.inf], np.nan)
        .fillna(0.0)
        .astype(np.float32)
    )


def safe_series(df, column):
    if column in df.columns:
        return numeric(df[column])

    return pd.Series(
        np.zeros(len(df), dtype=np.float32),
        index=df.index,
    )


def zero_series(df):
    return pd.Series(
        np.zeros(len(df), dtype=np.float32),
        index=df.index,
    )


def clean_text(value):
    if pd.isna(value):
        return ""

    return str(value).strip()


# ============================================================
# LABEL HARMONIZATION
# ============================================================

def map_unsw_label(attack_cat, binary_label):

    try:
        binary_label = int(float(binary_label))
    except Exception:
        binary_label = 1

    if binary_label == 0:
        return "Normal"

    label = clean_text(attack_cat).lower()

    if label == "normal":
        return "Normal"

    if label == "dos":
        return "DoS"

    if label == "reconnaissance":
        return "Reconnaissance"

    if label in {"backdoor", "backdoors"}:
        return "Backdoor"

    if label == "exploits":
        return "Exploits"

    if label == "fuzzers":
        return "Fuzzers"

    if label == "generic":
        return "Generic"

    # UNSW Analysis, Shellcode and Worms do not have a clean
    # one-to-one equivalent across all three datasets.
    return "OtherAttack"


def map_cic_label(value):

    label = clean_text(value).upper()

    if label == "BENIGN":
        return "Normal"

    if label.startswith("DDOS"):
        return "DDoS"

    if label.startswith("DOS"):
        return "DoS"

    if label.startswith("RECON"):
        return "Reconnaissance"

    if label == "VULNERABILITYSCAN":
        return "Reconnaissance"

    if label == "BACKDOOR_MALWARE":
        return "Backdoor"

    if label in {
        "SQLINJECTION",
        "COMMANDINJECTION",
        "XSS",
    }:
        return "Injection"

    if label == "DICTIONARYBRUTEFORCE":
        return "BruteForce"

    if label in {
        "MITM-ARPSPOOFING",
        "DNS_SPOOFING",
    }:
        return "MITM"

    if label.startswith("MIRAI"):
        return "Malware"

    if label in {
        "BROWSERHIJACKING",
        "UPLOADING_ATTACK",
    }:
        return "OtherAttack"

    return "OtherAttack"


def map_edge_label(value, binary_label=None):

    label = clean_text(value).lower()

    if binary_label is not None:
        try:
            if int(float(binary_label)) == 0:
                return "Normal"
        except Exception:
            pass

    if label == "normal":
        return "Normal"

    if label.startswith("ddos"):
        return "DDoS"

    if label in {
        "port_scanning",
        "fingerprinting",
        "vulnerability_scanner",
    }:
        return "Reconnaissance"

    if label == "backdoor":
        return "Backdoor"

    if label in {
        "sql_injection",
        "xss",
    }:
        return "Injection"

    if label == "password":
        return "BruteForce"

    if label == "mitm":
        return "MITM"

    if label == "ransomware":
        return "Malware"

    if label == "uploading":
        return "OtherAttack"

    return "OtherAttack"


# ============================================================
# UNSW FEATURE HARMONIZATION
# ============================================================

def harmonize_unsw_features(df):

    out = pd.DataFrame(index=df.index)

    out["duration"] = safe_series(df, "dur")

    total_packets = (
        safe_series(df, "spkts")
        + safe_series(df, "dpkts")
    )

    duration = safe_series(df, "dur")

    calculated_rate = np.where(
        duration.to_numpy() > 0,
        np.maximum(
            total_packets.to_numpy() - 1.0,
            0.0,
        )
        / duration.to_numpy(),
        0.0,
    )

    out["packet_rate"] = calculated_rate.astype(np.float32)

    out["src_packets"] = safe_series(df, "spkts")
    out["dst_packets"] = safe_series(df, "dpkts")

    out["src_bytes"] = safe_series(df, "sbytes")
    out["dst_bytes"] = safe_series(df, "dbytes")

    smean = safe_series(df, "smean")
    dmean = safe_series(df, "dmean")

    out["packet_size_mean"] = (
        (smean + dmean) / 2.0
    ).astype(np.float32)

    # Exact min/max/std packet size are not directly represented
    # in this raw UNSW schema.
    out["packet_size_min"] = zero_series(df)
    out["packet_size_max"] = zero_series(df)
    out["packet_size_std"] = zero_series(df)

    sttl = safe_series(df, "sttl")
    dttl = safe_series(df, "dttl")

    out["ttl"] = (
        (sttl + dttl) / 2.0
    ).astype(np.float32)

    # UNSW does not expose directly comparable per-flow TCP
    # flag-count fields in this representation.
    out["syn_count"] = zero_series(df)
    out["ack_count"] = zero_series(df)
    out["fin_count"] = zero_series(df)
    out["rst_count"] = zero_series(df)

    sinpkt = safe_series(df, "sinpkt")
    dinpkt = safe_series(df, "dinpkt")

    out["iat"] = (
        (sinpkt + dinpkt) / 2.0
    ).astype(np.float32)

    return out[COMMON_FEATURES]

def harmonize_cic_features(df):

    out = pd.DataFrame(index=df.index)

    # CICIoT2023 merged features do not expose a directly
    # comparable flow duration field here.
    out["duration"] = zero_series(df)

    out["packet_rate"] = safe_series(df, "Rate")

    # No directly equivalent directional packet counts.
    out["src_packets"] = zero_series(df)
    out["dst_packets"] = zero_series(df)

    # No directly equivalent directional byte totals.
    out["src_bytes"] = zero_series(df)
    out["dst_bytes"] = zero_series(df)

    out["packet_size_mean"] = safe_series(df, "AVG")
    out["packet_size_min"] = safe_series(df, "Min")
    out["packet_size_max"] = safe_series(df, "Max")
    out["packet_size_std"] = safe_series(df, "Std")

    out["ttl"] = safe_series(df, "Time_To_Live")

    out["syn_count"] = safe_series(df, "syn_count")
    out["ack_count"] = safe_series(df, "ack_count")
    out["fin_count"] = safe_series(df, "fin_count")
    out["rst_count"] = safe_series(df, "rst_count")

    out["iat"] = safe_series(df, "IAT")

    return out[COMMON_FEATURES]

# EDGE-IIoTset FEATURE HARMONIZATION
# ============================================================

def harmonize_edge_features(df):

    out = pd.DataFrame(index=df.index)

    # Edge-IIoTset DNN table is packet/protocol-oriented.
    # Many flow statistics used by UNSW/CIC are not directly
    # represented, so unsupported concepts remain zero.
    out["duration"] = zero_series(df)

    udp_delta = safe_series(df, "udp.time_delta")

    packet_rate = np.where(
        udp_delta.to_numpy() > 0,
        1.0 / udp_delta.to_numpy(),
        0.0,
    )

    # Avoid extreme reciprocal values.
    packet_rate = np.clip(
        packet_rate,
        0.0,
        1_000_000.0,
    )

    out["packet_rate"] = packet_rate.astype(np.float32)

    out["src_packets"] = zero_series(df)
    out["dst_packets"] = zero_series(df)

    out["src_bytes"] = zero_series(df)
    out["dst_bytes"] = zero_series(df)

    # tcp.len is the closest directly available packet/payload
    # length indicator in the selected DNN table.
    tcp_len = safe_series(df, "tcp.len")

    out["packet_size_mean"] = tcp_len
    out["packet_size_min"] = tcp_len
    out["packet_size_max"] = tcp_len
    out["packet_size_std"] = zero_series(df)

    out["ttl"] = zero_series(df)

    out["syn_count"] = safe_series(
        df,
        "tcp.connection.syn",
    )

    out["ack_count"] = safe_series(
        df,
        "tcp.flags.ack",
    )

    out["fin_count"] = safe_series(
        df,
        "tcp.connection.fin",
    )

    out["rst_count"] = safe_series(
        df,
        "tcp.connection.rst",
    )

    out["iat"] = udp_delta

    return out[COMMON_FEATURES]

def save_chunk(
    features,
    labels,
    dataset_name,
    chunk_number,
):

    features = features.copy()

    features.replace(
        [np.inf, -np.inf],
        np.nan,
        inplace=True,
    )

    features.fillna(0.0, inplace=True)

    for column in COMMON_FEATURES:
        features[column] = pd.to_numeric(
            features[column],
            errors="coerce",
        ).fillna(0.0).astype(np.float32)

    output = features.copy()

    output["label"] = labels.values
    output["dataset"] = dataset_name

    output_file = (
        OUTPUT_DIR
        / f"{dataset_name}_chunk_{chunk_number:04d}.csv"
    )

    output.to_csv(
        output_file,
        index=False,
    )

    return output_file

# Welford-style batch combination.

)

global_M2 = np.zeros(
    len(FEATURES),
    dtype=np.float64,
)

client_counts = {}

# ============================================================

for client in CLIENTS:

    section(f"PROCESSING TRAINING DATA: {client}")

    train_file = (
        DATA_DIR
        / client
        / "train_raw.csv"
    )

    if not train_file.exists():
        raise FileNotFoundError(
            f"Training file not found:\n{train_file}"
        )

    client_count = 0
    chunk_number = 0

    for chunk in pd.read_csv(
        train_file,
        usecols=FEATURES,
        chunksize=CHUNK_SIZE,
        low_memory=False,
    ):

        chunk_number += 1

        # ----------------------------------------------------
        # Convert safely to numeric
        # ----------------------------------------------------

        for feature in FEATURES:
            chunk[feature] = pd.to_numeric(
                chunk[feature],
                errors="coerce",
            )

        chunk = chunk.replace(
            [np.inf, -np.inf],
            np.nan,
        )

        # Harmonized data was already validated,
        # but keep this safety check.
        if chunk.isna().any().any():

            bad_count = int(
                chunk.isna().sum().sum()
            )

            raise ValueError(
                f"{client}: found {bad_count} "
                f"NaN/Infinity values in training data."
            )

        X = chunk.to_numpy(
            dtype=np.float64
        )

        n = X.shape[0]

        if n == 0:
            continue

        # ----------------------------------------------------
        # Batch statistics
        # ----------------------------------------------------

        batch_mean = np.mean(
            X,
            axis=0,
        )

        batch_M2 = np.sum(
            (X - batch_mean) ** 2,
            axis=0,
        )

        # ----------------------------------------------------
        # Merge batch stats into global stats
        # ----------------------------------------------------

        if global_count == 0:

            global_mean = batch_mean.copy()
            global_M2 = batch_M2.copy()
            global_count = n

        else:

            delta = (
                batch_mean
                - global_mean
            )

            new_count = (
                global_count
                + n
            )

            global_mean = (
                global_mean
                + delta
                * n
                / new_count
            )

            global_M2 = (
                global_M2
                + batch_M2
                + (delta ** 2)
                * global_count
                * n
                / new_count
            )

            global_count = new_count

        client_count += n

        print(
            f"Chunk {chunk_number:>3} | "
            f"Rows={n:>9,} | "
            f"Client total={client_count:>10,} | "
            f"Global total={global_count:>10,}"
        )

    )

        f"{client_count:,}"
    )

def section(title):

def reset_output_file(path):

    if path.exists():
        path.unlink()


def append_csv(df, path):

    header = not path.exists()

    df.to_csv(
        path,
        mode="a",
        header=header,
        index=False,
    )


# ============================================================
# DETERMINISTIC STRATIFIED SPLIT
# ============================================================
#
# We do not load the full 49.78M rows into memory.
#
# Each row receives a deterministic pseudo-random number based
# on:
#
#     dataset + source file + row number + class label + seed
#
# The hash is stable across runs.
#
# Because the same 80/20 rule is applied independently to
# every label, this approximates a stratified split while
# remaining streaming/chunk-friendly.
#
# ============================================================

def stable_train_mask(
    dataset_name,
    source_file,
    start_row,
    labels,
):

    mask = np.empty(
        len(labels),
        dtype=bool,
    )

    threshold = int(
        TRAIN_RATIO * 1_000_000
    )

    for i, label in enumerate(labels):

        row_number = start_row + i

        key = (
            f"{SEED}|"
            f"{dataset_name}|"
            f"{source_file}|"
            f"{row_number}|"
            f"{label}"
        )

        digest = hashlib.blake2b(
            key.encode("utf-8"),
            digest_size=8,
        ).digest()

        value = int.from_bytes(
            digest,
            byteorder="little",
            signed=False,
        )

        bucket = value % 1_000_000

        mask[i] = bucket < threshold

    return mask


# ============================================================
# RUNNING TRAIN STATISTICS
# ============================================================
#
# We calculate scaling statistics ONLY from training data.
#
# Uses:
#     count
#     sum
#     sum of squares
#
# Then:
#
# mean = sum / count
#
# variance =
#     (sum_of_squares / count) - mean^2
#
# std = sqrt(variance)
#
# ============================================================

class RunningStats:

    def __init__(self, n_features):

        self.n_features = n_features

        self.count = 0

        self.sum = np.zeros(
            n_features,
            dtype=np.float64,
        )

        self.sum_sq = np.zeros(
            n_features,
            dtype=np.float64,
        )


    def update(self, X):

        if len(X) == 0:
            return

        X = np.asarray(
            X,
            dtype=np.float64,
        )

        self.count += X.shape[0]

        self.sum += np.sum(
            X,
            axis=0,
        )

        self.sum_sq += np.sum(
            np.square(X),
            axis=0,
        )


    def finalize(self):

        if self.count == 0:

            raise ValueError(
                "No training records were found."
            )

        mean = self.sum / self.count

        variance = (
            self.sum_sq / self.count
        ) - np.square(mean)

        variance = np.maximum(
            variance,
            0.0,
        )

        std = np.sqrt(variance)

        # Prevent division by zero.
        std = np.where(
            std < 1e-12,
            1.0,
            std,
        )

        return mean, std

# PASS 1
# SPLIT DATA + CALCULATE TRAIN STATISTICS
# ============================================================

def split_dataset(
    dataset_name,
    config,
):

    section(
        f"CLIENT {config['client_id']} "
        f"- {dataset_name}"
    )

    files = sorted(
        INPUT_DIR.glob(
            config["pattern"]
        )
    )

    if not files:

        raise FileNotFoundError(
            f"No files found for {dataset_name}"
        )

    print(
        f"Source chunks: {len(files)}"
    )

    dataset_output = (
        OUTPUT_DIR
        / dataset_name
    )

    dataset_output.mkdir(
        parents=True,
        exist_ok=True,
    )

    train_raw_file = (
        dataset_output
        / "train_raw.csv"
    )

    test_raw_file = (
        dataset_output
        / "test_raw.csv"
    )

    reset_output_file(
        train_raw_file
    )

    reset_output_file(
        test_raw_file
    )

    stats = RunningStats(
        len(FEATURES)
    )

    total_rows = 0
    train_rows = 0
    test_rows = 0

    train_labels = Counter()
    test_labels = Counter()

    processed_files = 0

    required_columns = (
        FEATURES
        + ["label", "dataset"]
    )

    for path in files:

        file_row_offset = 0

        reader = pd.read_csv(
            path,
            chunksize=READ_CHUNK_SIZE,
            low_memory=False,
        )

        for df in reader:

            missing = [
                col
                for col in required_columns
                if col not in df.columns
            ]

            if missing:

                raise ValueError(
                    f"\nMissing columns in "
                    f"{path.name}:\n"
                    f"{missing}"
                )

            # ----------------------------------------
            # Validate dataset identity
            # ----------------------------------------

            dataset_values = (
                df["dataset"]
                .astype(str)
                .unique()
            )

            if (
                len(dataset_values) != 1
                or dataset_values[0]
                != dataset_name
            ):

                raise ValueError(
                    f"Dataset identity mismatch "
                    f"in {path.name}"
                )

            # ----------------------------------------
            # Validate labels
            # ----------------------------------------

            labels = (
                df["label"]
                .astype(str)
                .to_numpy()
            )

            invalid_labels = set(
                labels
            ) - set(
                CLASSES
            )

            if invalid_labels:

                raise ValueError(
                    f"Unexpected labels found:\n"
                    f"{invalid_labels}"
                )

            # ----------------------------------------
            # Numeric features
            # ----------------------------------------

            X = (
                df[FEATURES]
                .apply(
                    pd.to_numeric,
                    errors="coerce",
                )
                .replace(
                    [np.inf, -np.inf],
                    np.nan,
                )
                .fillna(0.0)
                .to_numpy(
                    dtype=np.float32
                )
            )

            # ----------------------------------------
            # Deterministic 80/20 split
            # ----------------------------------------

            train_mask = stable_train_mask(
                dataset_name,
                path.name,
                file_row_offset,
                labels,
            )

            test_mask = ~train_mask

            train_df = df.loc[
                train_mask,
                FEATURES + ["label"]
            ].copy()

            test_df = df.loc[
                test_mask,
                FEATURES + ["label"]
            ].copy()

            # Add numeric target.
            train_df["label_id"] = (
                train_df["label"]
                .map(LABEL_TO_ID)
                .astype(np.int16)
            )

            test_df["label_id"] = (
                test_df["label"]
                .map(LABEL_TO_ID)
                .astype(np.int16)
            )

            # ----------------------------------------
            # Save raw split
            # ----------------------------------------

            append_csv(
                train_df,
                train_raw_file,
            )

            append_csv(
                test_df,
                test_raw_file,
            )

            # ----------------------------------------
            # Training-only scaling statistics
            # ----------------------------------------

            stats.update(
                X[train_mask]
            )

            # ----------------------------------------
            # Counters
            # ----------------------------------------

            current_train_labels = Counter(
                labels[train_mask]
            )

            current_test_labels = Counter(
                labels[test_mask]
            )

            train_labels.update(
                current_train_labels
            )

            test_labels.update(
                current_test_labels
            )

            chunk_rows = len(df)

            total_rows += chunk_rows

            current_train_rows = int(
                np.sum(train_mask)
            )

            current_test_rows = int(
                np.sum(test_mask)
            )

            train_rows += (
                current_train_rows
            )

            test_rows += (
                current_test_rows
            )

            file_row_offset += (
                chunk_rows
            )

        processed_files += 1

        print(
            f"{dataset_name:<12} | "
            f"{processed_files:>3}/"
            f"{len(files):<3} chunks | "
            f"Total={total_rows:,} | "
            f"Train={train_rows:,} | "
            f"Test={test_rows:,}"
        )

    mean, std = stats.finalize()

    # ========================================================
    # SAVE SCALER
    # ========================================================

    scaler_file = (
        dataset_output
        / "scaler.npz"
    )

    np.savez(
        scaler_file,
        mean=mean.astype(np.float32),
        std=std.astype(np.float32),
        features=np.array(
            FEATURES,
            dtype=str,
        ),
    )

    # ========================================================
    # SAVE CLIENT METADATA
    # ========================================================

    metadata = {
        "dataset": dataset_name,
        "client_id": config["client_id"],
        "features": FEATURES,
        "classes": CLASSES,
        "label_to_id": LABEL_TO_ID,
        "total_rows": total_rows,
        "train_rows": train_rows,
        "test_rows": test_rows,
        "train_ratio_actual": (
            train_rows / total_rows
        ),
        "test_ratio_actual": (
            test_rows / total_rows
        ),
        "train_label_counts": dict(
            train_labels
        ),
        "test_label_counts": dict(
            test_labels
        ),
        "scaler_mean": mean.tolist(),
        "scaler_std": std.tolist(),
        "split_seed": SEED,
        "split_method": (
            "Deterministic streaming "
            "label-aware hash split"
        ),
        "important_note": (
            "Scaler statistics were calculated "
            "only from this client's training "
            "partition. Test records were not "
            "used to calculate scaling parameters."
        ),
    }

    metadata_file = (
        dataset_output
        / "client_metadata.json"
    )

    with open(
        metadata_file,
        "w",
        encoding="utf-8",
    ) as f:

        json.dump(
            metadata,
            f,
            indent=4,
        )

    print("\nSplit complete:")

    print(
        f"Total : {total_rows:,}"
    )

    print(
        f"Train : {train_rows:,} "
        f"({train_rows / total_rows * 100:.2f}%)"
    )

    print(
        f"Test  : {test_rows:,} "
        f"({test_rows / total_rows * 100:.2f}%)"
    )

    print("\nScaler saved:")
    print(scaler_file)

    print("\nMetadata saved:")
    print(metadata_file)

    return metadata

# VALIDATE CLASS COVERAGE
# ============================================================

def validate_class_coverage(
    metadata_list,
):

    section(
        "CLASS COVERAGE VALIDATION"
    )

    for metadata in metadata_list:

        dataset = metadata[
            "dataset"
        ]

        print(
            f"\n{dataset}"
        )

        print(
            "-" * 60
        )

        train_counts = metadata[
            "train_label_counts"
        ]

        test_counts = metadata[
            "test_label_counts"
        ]

        dataset_classes = sorted(
            set(train_counts.keys())
            | set(test_counts.keys())
        )

        for label in dataset_classes:

            train_count = int(
                train_counts.get(
                    label,
                    0,
                )
            )

            test_count = int(
                test_counts.get(
                    label,
                    0,
                )
            )

            status = (
                "PASS"
                if (
                    train_count > 0
                    and test_count > 0
                )
                else "CHECK"
            )

            print(
                f"{label:<20} "
                f"Train={train_count:>10,} | "
                f"Test={test_count:>9,} | "
                f"{status}"
            )

def save_global_metadata(
    metadata_list,
):

    section(
        "GLOBAL FEDERATED DATA SUMMARY"
    )

    total_rows = sum(
        m["total_rows"]
        for m in metadata_list
    )

    total_train = sum(
        m["train_rows"]
        for m in metadata_list
    )

    total_test = sum(
        m["test_rows"]
        for m in metadata_list
    )

    print(
        f"Total records : "
        f"{total_rows:,}"
    )

    print(
        f"Train records : "
        f"{total_train:,}"
    )

    print(
        f"Test records  : "
        f"{total_test:,}"
    )

    print("\nFederated clients:")

    for metadata in metadata_list:

        print(
            f"Client "
            f"{metadata['client_id']} -> "
            f"{metadata['dataset']} | "
            f"Train="
            f"{metadata['train_rows']:,} | "
            f"Test="
            f"{metadata['test_rows']:,}"
        )

    global_metadata = {
        "federated_setup": (
            "3-client cross-dataset "
            "non-IID simulated federated learning"
        ),
        "number_of_clients": 3,
        "features": FEATURES,
        "number_of_features": len(
            FEATURES
        ),
        "classes": CLASSES,
        "number_of_classes": len(
            CLASSES
        ),
        "label_to_id": LABEL_TO_ID,
        "total_records": total_rows,
        "total_train_records": total_train,
        "total_test_records": total_test,
        "clients": [
            {
                "client_id": m[
                    "client_id"
                ],
                "dataset": m[
                    "dataset"
                ],
                "total_rows": m[
                    "total_rows"
                ],
                "train_rows": m[
                    "train_rows"
                ],
                "test_rows": m[
                    "test_rows"
                ],
            }
            for m in metadata_list
        ],
        "methodology_note": (
            "The three datasets are retained "
            "as separate heterogeneous federated "
            "clients rather than concatenated into "
            "one centralized training dataset."
        ),
    }

    path = (
        OUTPUT_DIR
        / "global_metadata.json"
    )

    with open(
        path,
        "w",
        encoding="utf-8",
    ) as f:

        json.dump(
            global_metadata,
            f,
            indent=4,
        )

    print("\nGlobal metadata saved:")
    print(path)
